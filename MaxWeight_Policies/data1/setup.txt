function [K,vector_A,num_iterations,set_simplified,optimal_cost_Linear_Max_Weight,optimal_cost_Buffered_Max_Weight,optimal_cost_LowerBound_stoch,optimal_cost_Throughput_Max_Weight,optimal_cost_Rand]=Simulation_varying_A
%[K,vector_A,num_iterations,set_simplified,optimal_cost_Linear_Max_Weight,optimal_cost_Buffered_Max_Weight,optimal_cost_LowerBound_stoch,optimal_cost_Throughput_Max_Weight,optimal_cost_Rand]=Simulation_varying_A
%[K,vector_A,num_iterations,set_simplified,optimal_cost_Linear_Max_Weight,optimal_cost_Quadratic_Max_Weight,optimal_cost_LowerBound_stoch,optimal_cost_Whittle_Single_Buffer,optimal_cost_Rand,optimal_cost_Whittle_NO_Buffer,throughput_Linear_Max_Weight,throughput_Quadratic_Max_Weight,throughput_Whittle_Single_Buffer,throughput_Rand,throughput_Whittle_NO_Buffer]=Simulation_varying_A
%[K,vector_M,num_iterations,set_simplified,optimal_cost_Linear_Max_Weight,optimal_cost_Quadratic_Max_Weight,optimal_cost_LowerBound_stoch,optimal_cost_Whittle_Single_Buffer_Throughput_Constraint,average_throughput_Linear_Max_Weight,average_throughput_Quadratic_Max_Weight,average_throughput_Whittle_Single_Buffer]=Simulation_varying_A

clear
clc

global probability cost alpha arrival optQ optMU

tic;

%% Model Setup
M=2;
vector_A=0.01:0.01:1;
%epsilon=0.9;
%vector_M=[2,4,6];
num_setups=length(vector_A);
num_iterations=10;
%num_iterations=3;
T=1;
set_simplified=1;
h1=1; %initial age
z1=0; %HAS TO BE LOWER THAN H1

%% Parameters Setup
optimal_cost_Linear_Max_Weight=zeros(1,num_setups);
%optimal_cost_Quadratic_Max_Weight=zeros(1,num_setups);
optimal_cost_LowerBound_stoch=zeros(1,num_setups);
%optimal_cost_Whittle_Single_Buffer=zeros(1,num_setups);
optimal_cost_Rand=zeros(1,num_setups);
%optimal_cost_Whittle_NO_Buffer=zeros(1,num_setups);
optimal_cost_Buffered_Max_Weight=zeros(1,num_setups);
optimal_cost_Throughput_Max_Weight=zeros(1,num_setups);

%throughput_Linear_Max_Weight=zeros(max(vector_A),num_setups);
%throughput_Quadratic_Max_Weight=zeros(max(vector_A),num_setups);
%throughput_Whittle_Single_Buffer=zeros(max(vector_A),num_setups);
%throughput_Rand=zeros(max(vector_A),num_setups);
%throughput_Whittle_NO_Buffer=zeros(max(vector_M),num_setups);
%throughput_Buffered_Max_Weight=zeros(max(vector_M),num_setups);
%throughput_Throughput_Max_Weight=zeros(max(vector_M),num_setups);


%% Simulations
for count=1:num_setups
    count    
    K=1000000*M;
    %K=100*M;    
    cost=zeros(M,1);
    alpha=ones(M,1);
    arrival=vector_A(count)*[1;1/3];
    probability=[1/3;1];
        
    Lower_Bound_find_q(M); %findind the optQ            
    optimal_cost_LowerBound_stoch(count)=sum(alpha.*(1+1./optQ))/(2*M);
    disp('Lower Bound for Stochastic Arrivals')         
    
    optMU=sqrt(alpha./probability);
    sum_optMU=sum(optMU);
    optMU=optMU/sum_optMU;
    
    set_Policy=1; %LINEAR MaxWeight    
    [aux,optimal_cost_Linear_Max_Weight(count)]=MaxWeight_and_Greedy_simulation_stochastic(K,T,M,h1,z1,num_iterations,set_Policy);    
    optimal_cost_Linear_Max_Weight(count)=optimal_cost_Linear_Max_Weight(count)/(M*T);
    %average_throughput_Linear_Max_Weight(count)=mean(throughput_Linear_Max_Weight);    
    disp('Linear Max-Weight policy')
    
%     set_Policy=2; %Quadratic MaxWeight    
%     [throughput_Quadratic_Max_Weight(1:M,count),optimal_cost_Quadratic_Max_Weight(count)]=MaxWeight_and_Greedy_simulation_stochastic(K,T,M,h1,z1,num_iterations,set_Policy);
%     optimal_cost_Quadratic_Max_Weight(count)=optimal_cost_Quadratic_Max_Weight(count)/(M*T);
%     %average_throughput_Quadratic_Max_Weight(count)=mean(throughput_Quadratic_Max_Weight);    
%     disp('Quadratic Max-Weight policy')    
    
    work_conserving_policy=0; %NOT Work Conserving POLICY
    [aux,optimal_cost_Rand(count)]=Random_simulation_stochastic(K,T,M,h1,z1,num_iterations,work_conserving_policy);    
    optimal_cost_Rand(count)=optimal_cost_Rand(count)/(M*T);
    %average_throughput_Rand(count)=mean(throughput_Rand);
    disp('Randomized policy')
    
%     set_Policy=1; %Whittle with NO BUFFER (Yu-Pin) case
%     [throughput_Whittle_NO_Buffer(1:M,count),optimal_cost_Whittle_NO_Buffer(count)]=Whittle_simulation_stochastic(K,T,M,h1,z1,num_iterations,set_Policy);
%     optimal_cost_Whittle_NO_Buffer(count)=optimal_cost_Whittle_NO_Buffer(count)/(M*T);
%     %average_throughput_Whittle_NO_Buffer(count)=mean(throughput_Whittle_NO_Buffer);
%     disp('Whittle Index policy for No Buffer case')
%     
%     set_Policy=0; %Whittle with Single Buffer that keeps freshest packet
%     [throughput_Whittle_Single_Buffer(1:M,count),optimal_cost_Whittle_Single_Buffer(count)]=Whittle_simulation_stochastic(K,T,M,h1,z1,num_iterations,set_Policy);
%     optimal_cost_Whittle_Single_Buffer(count)=optimal_cost_Whittle_Single_Buffer(count)/(M*T);
%     %average_throughput_Whittle_Single_Buffer(count)=mean(throughput_Whittle_Single_Buffer);
%     disp('Whittle Index policy for the Freshest Packet case')
    
    set_Policy=1; %Buffered MaxWeight    
    [aux,optimal_cost_Buffered_Max_Weight(count)]=MaxWeight_Buffered_simulation_stochastic(K,T,M,h1,z1,num_iterations,set_Policy);    
    optimal_cost_Buffered_Max_Weight(count)=optimal_cost_Buffered_Max_Weight(count)/(M*T);
    %average_throughput_Linear_Max_Weight(count)=mean(throughput_Linear_Max_Weight);    
    disp('Buffered Max-Weight policy')
    
    set_Policy=0; %Throughput MaxWeight    
    [aux,optimal_cost_Throughput_Max_Weight(count)]=MaxWeight_Buffered_simulation_stochastic(K,T,M,h1,z1,num_iterations,set_Policy);    
    optimal_cost_Throughput_Max_Weight(count)=optimal_cost_Throughput_Max_Weight(count)/(M*T);
    %average_throughput_Linear_Max_Weight(count)=mean(throughput_Linear_Max_Weight);    
    disp('Throughput Max-Weight policy')
        
    toc
end

figure(3)
[Linear_Max_Weight_plot]=plot(vector_A,optimal_cost_Linear_Max_Weight,'bo--','LineWidth',3,'MarkerSize',20);
hold on
%[Quadratic_Max_Weight_plot]=plot(vector_M,optimal_cost_Quadratic_Max_Weight,'rv--','LineWidth',3,'MarkerSize',20);
[Buffered_Max_Weight_plot]=plot(vector_A,optimal_cost_Buffered_Max_Weight,'rv--','LineWidth',3,'MarkerSize',20);
%[Whittle_Single_Buffer_plot]=plot(vector_M,optimal_cost_Whittle_Single_Buffer,'m+--','LineWidth',3,'MarkerSize',25);
[Throughput_Max_Weight_plot]=plot(vector_A,optimal_cost_Throughput_Max_Weight,'m+--','LineWidth',3,'MarkerSize',25);
[Rand_plot]=plot(vector_A,optimal_cost_Rand,'gx--','LineWidth',3,'MarkerSize',25);
%[Whittle_NO_Buffer_plot]=plot(vector_M,optimal_cost_Whittle_NO_Buffer,'k^--','LineWidth',3,'MarkerSize',20);
[Lower_Bound]=plot(vector_A,optimal_cost_LowerBound_stoch,'k-','LineWidth',3,'MarkerSize',20);
%legend([Linear_Max_Weight_plot,Quadratic_Max_Weight_plot,Whittle_Single_Buffer_plot,Lower_Bound,Rand_plot,Whittle_NO_Buffer_plot],'Linear Max-Weight','Quadratic Max-Weight','Whittle Index','Lower Bound','Stationary Randomized','Whittle with no buffer','Location','NorthWest');
legend([Linear_Max_Weight_plot,Buffered_Max_Weight_plot,Throughput_Max_Weight_plot,Lower_Bound,Rand_plot],'Linear Max-Weight','Buffered Max-Weight','Throughput Max-Weight','Lower Bound','Stationary Randomized','Location','NorthWest');
ylabel('Expected Weighted Sum AoI')
xlabel('Arrival Probability, A')
%title('Comparing the performance of different policies')
hold off

toc;
end
