function [K,vector_A,num_iterations,set_simplified,optimal_cost_Max_Weight,optimal_cost_LowerBound_stoch,optimal_cost_Whittle,optimal_cost_Rand,optimal_cost_Rand_FIFO,optimal_cost_Whittle_NO_Buffer,optimal_cost_Max_Weight_FIFO,optimal_cost_Rand_NO_Buffer,optimal_cost_Throughput_Max_Weight_FIFO,optimal_cost_Max_Weight_NO_buff]=Simulation_varying_A
%[K,vector_A,num_iterations,set_simplified,optimal_cost_Linear_Max_Weight,optimal_cost_Buffered_Max_Weight,optimal_cost_LowerBound_stoch,optimal_cost_Throughput_Max_Weight,optimal_cost_Rand]=Simulation_varying_A
%[K,vector_A,num_iterations,set_simplified,optimal_cost_Linear_Max_Weight,optimal_cost_Quadratic_Max_Weight,optimal_cost_LowerBound_stoch,optimal_cost_Whittle_Single_Buffer,optimal_cost_Rand,optimal_cost_Whittle_NO_Buffer,throughput_Linear_Max_Weight,throughput_Quadratic_Max_Weight,throughput_Whittle_Single_Buffer,throughput_Rand,throughput_Whittle_NO_Buffer]=Simulation_varying_A
%[K,vector_M,num_iterations,set_simplified,optimal_cost_Linear_Max_Weight,optimal_cost_Quadratic_Max_Weight,optimal_cost_LowerBound_stoch,optimal_cost_Whittle_Single_Buffer_Throughput_Constraint,average_throughput_Linear_Max_Weight,average_throughput_Quadratic_Max_Weight,average_throughput_Whittle_Single_Buffer]=Simulation_varying_A

clear
clc

global probability cost alpha arrival optQ optMU optMU_buff optMU_NO_buff

tic;

%% Model Setup
M=4;
vector_A=0.01:0.01:0.4;
%epsilon=0.9;
%vector_M=[2,4,6];
num_setups=length(vector_A);
%num_iterations=10;
num_iterations=10;
T=1;
set_simplified=1;
h1=1; %initial age
z1=0; %HAS TO BE LOWER THAN H1

%% Parameters Setup
optimal_cost_Rand=zeros(1,num_setups);
optimal_cost_Rand_FIFO=zeros(1,num_setups);
optimal_cost_Rand_NO_Buffer=zeros(1,num_setups);
optimal_cost_Whittle_NO_Buffer=zeros(1,num_setups);
optimal_cost_Whittle=zeros(1,num_setups);
optimal_cost_Max_Weight_FIFO=zeros(1,num_setups);
optimal_cost_Max_Weight_NO_buff=zeros(1,num_setups);
optimal_cost_Max_Weight=zeros(1,num_setups);
optimal_cost_Throughput_Max_Weight_FIFO=zeros(1,num_setups);
optimal_cost_LowerBound_stoch=zeros(1,num_setups);

%throughput_Linear_Max_Weight=zeros(max(vector_A),num_setups);
%throughput_Quadratic_Max_Weight=zeros(max(vector_A),num_setups);
%throughput_Whittle_Single_Buffer=zeros(max(vector_A),num_setups);
%throughput_Rand=zeros(max(vector_A),num_setups);
%throughput_Whittle_NO_Buffer=zeros(max(vector_M),num_setups);
%throughput_Buffered_Max_Weight=zeros(max(vector_M),num_setups);
%throughput_Throughput_Max_Weight=zeros(max(vector_M),num_setups);


%% Simulations
for count=1:num_setups
    count    
    %K=1000000*M;
    K=500000*M; 
    cost=zeros(M,1);
    alpha=[4;4;1;1];
    arrival=vector_A(count)*[1;3/4;1/2;1/4];
    probability=[1/4;1/2;3/4;1];
    
    p=zeros(M,1);
    p=p+probability; %Making sure that "probability" has the right size

    A=zeros(M,1);
    A =A+alpha; %Making sure that "alpha" has the right size
    
    Arr=zeros(M,1);
    Arr=Arr+arrival; %Making sure that "constraint" has the right size
    
    Lower_Bound_find_q(M); %findind the optQ            
    optimal_cost_LowerBound_stoch(count)=sum(alpha.*(1+1./optQ))/(2*M);
    disp('Lower Bound for Stochastic Arrivals')         
    
    optMU=sqrt(alpha./probability);
    sum_optMU=sum(optMU);
    optMU=optMU/sum_optMU;
        
    optMU_NO_buff=sqrt(alpha./(arrival.*probability));
    sum_optMU_NO_buff=sum(optMU_NO_buff);
    optMU_NO_buff=optMU_NO_buff/sum_optMU_NO_buff;
    
    Random_find_mu_buffered_BerBer1(M) %findind the optMU_buff   
    optimal_cost_Rand(count)=(sum(sqrt(A./p)))^2/M+(sum(A.*(1./Arr-1)))/M;
    optimal_cost_Rand_NO_Buffer(count)=(sum(sqrt(A./(p.*Arr))))^2/M;
    if sum(Arr./p)>=1
        optimal_cost_Rand_FIFO(count)=1e10;        
    else
        optimal_cost_Rand_FIFO(count)=(sum(A.*(1./Arr+1./(p.*optMU_buff)+(Arr./(p.*optMU_buff)).^2.*((1-p.*optMU_buff)./(p.*optMU_buff-Arr)))))/M;        
        optMU_buff        
    end     
    
    set_Policy=1; %LINEAR MaxWeight    
    [aux,optimal_cost_Max_Weight(count)]=MaxWeight_and_Greedy_simulation_stochastic(K,T,M,h1,z1,num_iterations,set_Policy);    
    optimal_cost_Max_Weight(count)=optimal_cost_Max_Weight(count)/(M*T);
    %average_throughput_Linear_Max_Weight(count)=mean(throughput_Linear_Max_Weight);    
    disp('Linear Max-Weight policy with Optimal Buffer')
    
    set_Policy=3; %LINEAR MaxWeight with NO BUFFER
    [aux,optimal_cost_Max_Weight_NO_buff(count)]=MaxWeight_and_Greedy_simulation_stochastic(K,T,M,h1,z1,num_iterations,set_Policy);    
    optimal_cost_Max_Weight_NO_buff(count)=optimal_cost_Max_Weight_NO_buff(count)/(M*T);
    %average_throughput_Linear_Max_Weight(count)=mean(throughput_Linear_Max_Weight);    
    disp('Linear Max-Weight policy with no Buffer')
    
%     set_Policy=2; %Quadratic MaxWeight    
%     [throughput_Quadratic_Max_Weight(1:M,count),optimal_cost_Quadratic_Max_Weight(count)]=MaxWeight_and_Greedy_simulation_stochastic(K,T,M,h1,z1,num_iterations,set_Policy);
%     optimal_cost_Quadratic_Max_Weight(count)=optimal_cost_Quadratic_Max_Weight(count)/(M*T);
%     %average_throughput_Quadratic_Max_Weight(count)=mean(throughput_Quadratic_Max_Weight);    
%     disp('Quadratic Max-Weight policy')    
    
    %work_conserving_policy=0; %NOT Work Conserving POLICY
    %[aux,optimal_cost_Rand(count)]=Random_simulation_stochastic(K,T,M,h1,z1,num_iterations,work_conserving_policy);    
    %optimal_cost_Rand(count)=optimal_cost_Rand(count)/(M*T);
    %%average_throughput_Rand(count)=mean(throughput_Rand);
    %disp('Stationary Policy with Optimal Buffer')
    
    set_Policy=1; %Whittle with NO BUFFER (Yu-Pin) case
    [aux,optimal_cost_Whittle_NO_Buffer(count)]=Whittle_simulation_stochastic(K,T,M,h1,z1,num_iterations,set_Policy);
    optimal_cost_Whittle_NO_Buffer(count)=optimal_cost_Whittle_NO_Buffer(count)/(M*T);
    %average_throughput_Whittle_NO_Buffer(count)=mean(throughput_Whittle_NO_Buffer);
    disp('Whittle Index policy for No Buffer case')
     
    %set_Policy=0; %Whittle with Single Buffer that keeps freshest packet
    %[aux,optimal_cost_Whittle(count)]=Whittle_simulation_stochastic(K,T,M,h1,z1,num_iterations,set_Policy);
    %optimal_cost_Whittle(count)=optimal_cost_Whittle(count)/(M*T);
    %average_throughput_Whittle_Single_Buffer(count)=mean(throughput_Whittle_Single_Buffer);
    %disp('Whittle Index policy for the Freshest Packet case')

    set_Policy=1; %Buffered MaxWeight    
    [aux,optimal_cost_Max_Weight_FIFO(count)]=MaxWeight_Buffered_simulation_stochastic(K,T,M,h1,z1,num_iterations,set_Policy);    
    optimal_cost_Max_Weight_FIFO(count)=optimal_cost_Max_Weight_FIFO(count)/(M*T);
    %average_throughput_Linear_Max_Weight(count)=mean(throughput_Linear_Max_Weight);    
    disp('Linear Max-Weight policy with FIFO Buffer')
    
    %set_Policy=0; %Throughput MaxWeight    
    %[aux,optimal_cost_Throughput_Max_Weight_FIFO(count)]=MaxWeight_Buffered_simulation_stochastic(K,T,M,h1,z1,num_iterations,set_Policy);    
    %optimal_cost_Throughput_Max_Weight_FIFO(count)=optimal_cost_Throughput_Max_Weight_FIFO(count)/(M*T);
    %average_throughput_Linear_Max_Weight(count)=mean(throughput_Linear_Max_Weight);    
    %disp('Throughput Max-Weight policy with FIFO Buffer')
        
    toc
end

figure(3)
[Rand_FIFO_plot]=plot(vector_A,optimal_cost_Rand_FIFO,'mo--','LineWidth',3,'MarkerSize',20);
hold on
[Rand_No_Buff_plot]=plot(vector_A,optimal_cost_Rand_NO_Buffer,'ro--','LineWidth',3,'MarkerSize',20);
[Rand_plot]=plot(vector_A,optimal_cost_Rand,'bo--','LineWidth',3,'MarkerSize',20);
[Buffered_Max_Weight_plot]=plot(vector_A,optimal_cost_Max_Weight_FIFO,'mx--','LineWidth',3,'MarkerSize',25);
[Linear_Max_Weight_No_buff_plot]=plot(vector_A,optimal_cost_Max_Weight_NO_buff,'rx--','LineWidth',3,'MarkerSize',25);
[Linear_Max_Weight_plot]=plot(vector_A,optimal_cost_Max_Weight,'bx--','LineWidth',3,'MarkerSize',25);
[Whittle_NO_Buffer_plot]=plot(vector_A,optimal_cost_Whittle_NO_Buffer,'r^--','LineWidth',3,'MarkerSize',20);
[Whittle_Single_Buffer_plot]=plot(vector_A,optimal_cost_Whittle,'b^--','LineWidth',3,'MarkerSize',20);
[Throughput_Max_Weight_plot]=plot(vector_A,optimal_cost_Throughput_Max_Weight_FIFO,'g+--','LineWidth',3,'MarkerSize',25);
[Lower_Bound]=plot(vector_A,optimal_cost_LowerBound_stoch,'k-','LineWidth',3,'MarkerSize',20);
legend([Rand_FIFO_plot,Rand_No_Buff_plot,Rand_plot,Buffered_Max_Weight_plot,Linear_Max_Weight_No_buff_plot,Linear_Max_Weight_plot,Whittle_NO_Buffer_plot,Whittle_Single_Buffer_plot,Throughput_Max_Weight_plot,Lower_Bound],'Randomized with FIFO Buffer','Randomized with No Buffer','Randomized with Optimal Buffer','Max-Weight with FIFO Buffer','Max-Weight with No Buffer','Max-Weight with Optimal Buffer','Whittle with No Buffer','Whittle with Optimal Buffer','Max-Weight for Backlog','Lower Bound','Location','NorthWest');
ylabel('Expected Weighted Sum AoI')
xlabel('Arrival Probability, A')
%title('Comparing the performance of different policies')
hold off

toc;
end
